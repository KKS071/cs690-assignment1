def top_k_frequent(items, k):
    if k == 0:
        return []

    counts = {}
    first_seen = {}

    for index, item in enumerate(items):
        if item not in counts:
            counts[item] = 0
            first_seen[item] = index
        counts[item] += 1

    ordered = sorted(counts, key=lambda item: (-counts[item], first_seen[item]))
    return ordered[:k]
